        function getGradeInfo(total) {
            if (total >= 40) return { grade: 'A', title: 'GRED A (CEMERLANG)', badgeClass: 'bg-emerald-100 text-emerald-800 border-emerald-300', barClass: 'bg-emerald-500' };
            if (total >= 35) return { grade: 'B', title: 'GRED B (KEPUASAN)', badgeClass: 'bg-blue-100 text-blue-800 border-blue-300', barClass: 'bg-blue-500' };
            if (total >= 30) return { grade: 'C', title: 'GRED C (LULUS)', badgeClass: 'bg-amber-100 text-amber-800 border-amber-300', barClass: 'bg-amber-500' };
            if (total >= 25) return { grade: 'D', title: 'GRED D (MEMUASKAN)', badgeClass: 'bg-orange-100 text-orange-800 border-orange-300', barClass: 'bg-orange-500' };
            return { grade: 'F', title: 'GRED F (GAGAL)', badgeClass: 'bg-red-100 text-red-800 border-red-300', barClass: 'bg-red-500' };
        }

        function handleSearch() {
            const query = document.getElementById('searchInput').value.trim().toUpperCase();
            const clearBtn = document.getElementById('clearBtn');
            const suggestionsEl = document.getElementById('suggestions');
            const emptyState = document.getElementById('emptyState');
            const studentCard = document.getElementById('studentCard');
            const multipleResultsList = document.getElementById('multipleResultsList');

            if (query.length > 0) {
                clearBtn.classList.remove('hidden');
            } else {
                clearBtn.classList.add('hidden');
                suggestionsEl.classList.add('hidden');
                emptyState.classList.remove('hidden');
                studentCard.classList.add('hidden');
                multipleResultsList.classList.add('hidden');
                return;
            }

            // Filter database matching Matrik or Name
            const matches = studentDatabase.filter(s => 
                s.matrik.toUpperCase().includes(query) || 
                s.nama.toUpperCase().includes(query)
            );

            // Populate suggestions dropdown
            if (matches.length > 0 && query.length >= 2) {
                suggestionsEl.innerHTML = matches.slice(0, 6).map(s => `
                    <div onclick="selectStudent('${s.matrik}')" class="p-3 hover:bg-slate-100 cursor-pointer border-b border-slate-100 flex items-center justify-between text-xs sm:text-sm transition">
                        <div>
                            <span class="font-bold text-navy-900">${s.nama}</span>
                            <span class="text-xs text-slate-500 block font-mono">${s.matrik} • ${s.praktikum}</span>
                        </div>
                        <span class="bg-navy-100 text-navy-800 px-2 py-0.5 rounded font-semibold text-xs">${s.set}</span>
                    </div>
                `).join('');
                suggestionsEl.classList.remove('hidden');
            } else {
                suggestionsEl.classList.add('hidden');
            }

            if (matches.length === 1) {
                renderStudentCard(matches[0]);
            } else if (matches.length > 1) {
                renderMultipleResults(matches);
            } else {
                renderNotFound();
            }
        }

        function selectStudent(matrik) {
            document.getElementById('suggestions').classList.add('hidden');
            const student = studentDatabase.find(s => s.matrik === matrik);
            if (student) {
                document.getElementById('searchInput').value = student.matrik;
                renderStudentCard(student);
            }
        }

        function clearSearch() {
            document.getElementById('searchInput').value = '';
            handleSearch();
            document.getElementById('searchInput').focus();
        }

        function openPdfModal() {
            if (!activeStudent) return;

            const modal = document.getElementById('pdfModal');
            const gradeInfo = getGradeInfo(activeStudent.total);

            // Set modal dynamic content
            document.getElementById('modalPdfTitle').textContent = activeStudent.pdf;
            document.getElementById('modalStudentMeta').textContent = `${activeStudent.nama} (${activeStudent.matrik})`;
            
            document.getElementById('pdfStudentName').textContent = activeStudent.nama;
            document.getElementById('pdfStudentMatrik').textContent = activeStudent.matrik;
            document.getElementById('pdfStudentPraktikum').textContent = activeStudent.praktikum;
            document.getElementById('pdfStudentSet').textContent = activeStudent.set;
            document.getElementById('pdfStudentLecturer').textContent = activeStudent.pensyarah;

            document.getElementById('pdfScoreTI').textContent = activeStudent.ti.toFixed(1);
            document.getElementById('pdfScoreNum').textContent = activeStudent.num.toFixed(1);
            document.getElementById('pdfScoreTotal').textContent = activeStudent.total.toFixed(1);

            document.getElementById('pdfGradeTitle').textContent = gradeInfo.title;
            document.getElementById('pdfGradeBadge').textContent = gradeInfo.grade;
            document.getElementById('pdfFileName').textContent = activeStudent.pdf;

            // Set today date
            const today = new Date();
            document.getElementById('pdfCurrentDate').textContent = today.toLocaleDateString('ms-MY');

            modal.classList.remove('hidden');
            document.body.style.overflow = 'hidden';
        }

        function closePdfModal() {
            document.getElementById('pdfModal').classList.add('hidden');
            document.body.style.overflow = 'auto';
        }

        // PDF Actions Simulation
        function downloadPdfFile() {
            if (!activeStudent) return;
            
            // Create dummy blob for simulated pdf download
            const element = document.createElement('a');
            const fileContent = `=== KOLEJ MATRIKULASI MALAYSIA ===\nSLIP LAPORAN MARKAH PELAJAR\n\nNama: ${activeStudent.nama}\nNo Matrik: ${activeStudent.matrik}\nPensyarah: ${activeStudent.pensyarah}\nPraktikum: ${activeStudent.praktikum}\nSet Soalan: ${activeStudent.set}\n\nMarkah TI (/45): ${activeStudent.ti}\nNumerical Skill (/5): ${activeStudent.num}\nJUMLAH MARKAH (/50): ${activeStudent.total}\n\nFail Asal: ${activeStudent.pdf}\nDisahkan Rasmi oleh Jabatan Matematik.`;
            
            const file = new Blob([fileContent], { type: 'text/plain;charset=utf-8' });
            element.href = URL.createObjectURL(file);
            element.download = activeStudent.pdf.endsWith('.pdf') ? activeStudent.pdf : activeStudent.pdf + '.pdf';
            document.body.appendChild(element);
            element.click();
            document.body.removeChild(element);
        }

        }

        function openPdfNewTab() {
            if (!activeStudent) return;
            const fileContent = `=== KOLEJ MATRIKULASI MALAYSIA ===%0ANama: ${encodeURIComponent(activeStudent.nama)}%0ANo Matrik: ${activeStudent.matrik}%0AJumlah Markah: ${activeStudent.total}/50`;
            window.open(`data:text/plain;charset=utf-8,${fileContent}`, '_blank');
        }
